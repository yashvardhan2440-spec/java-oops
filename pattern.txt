System.out.print("Enter n: ");
    int n = sc.nextInt();

    int rows = 2 * n - 1;

    for (int i = 1; i <= rows; i++) {

        int dots;
        if (i <= n) {
            dots = i;
        } else {
            dots = rows - i + 1;
        }


        for (int j = 1; j <= n - dots; j++) {
            System.out.print("  ");
        }

        for (int j = 1; j <= dots; j++) {
            System.out.print("*");
            if (j != dots) {
                System.out.print("   ");
            }
        }

        System.out.println();
    }

    sc.close();
}
